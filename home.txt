<script>
(function () {
  var page = document.querySelector('.nh-page');
  if (!page) return;
  var btns = page.querySelectorAll('[data-nh-filter]');
  var cards = page.querySelectorAll('[data-nh-tags]');
  var empty = page.querySelector('[data-nh-empty]');
  Array.prototype.forEach.call(btns, function (b) {
    b.addEventListener('click', function () {
      var f = b.getAttribute('data-nh-filter');
      var shown = 0;
      Array.prototype.forEach.call(btns, function (x) {
        var on = x === b;
        x.setAttribute('aria-pressed', on ? 'true' : 'false');
        x.style.background = on ? '#0F3D6B' : '#F1E9D6';
        x.style.borderColor = on ? '#0F3D6B' : '#CDBF9E';
        x.style.color = on ? '#FFFFFF' : '#06172E';
        x.style.fontWeight = on ? '500' : '400';
      });
      Array.prototype.forEach.call(cards, function (c) {
        var tags = ' ' + c.getAttribute('data-nh-tags') + ' ';
        var ok = f === 'all' || tags.indexOf(' ' + f + ' ') >= 0;
        c.style.display = ok ? 'flex' : 'none';
        if (ok) shown++;
      });
      if (empty) empty.style.display = shown ? 'none' : 'block';
    });
  });
})();
</script>
